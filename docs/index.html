Developer First

Simple REST API

Submit a URL, get back a detection report. That's it.

Python 
import requests

# Submit a detection job
response = requests.post(
    "https://api.aiblock.dev/v1/detect",
    headers={"Authorization": "Bearer YOUR_API_KEY"},
    json={
        "url": "https://www.youtube.com/watch?v=dQw4w9WgXcQ",
        "types": ["audio", "video"]
    }
)

job = response.json()
# {"job_id": "det_abc123", "status": "queued"}

# Fetch results
result = requests.get(
    f"https://api.aiblock.dev/v1/jobs/{job['job_id']}",
    headers={"Authorization": "Bearer YOUR_API_KEY"}
).json()

print(result["detection"])
# {"audio": {"ai_probability": 0.94, "model": "ElevenLabs", ...},
#  "video": {"deepfake_score": 0.12, "confidence": 0.97}}
